# Plan 03-01: WaveSurfer Waveform Panel, Playback Store & Region Sync

## Objective

Install WaveSurfer packages, create `usePlaybackStore` managing WaveSurfer instance reference, current time, and word selection state (anchor + end indices). Build `WaveformPanel.vue` using `useWaveSurfer` composable to display the recorded audio waveform, expose a Regions plugin region that reflects the selected word range, and push `currentTime` events into the playback store for cross-component access.

## Must-Haves

1. WaveSurfer renders the audio waveform when `audioBlobUrl` is available
2. `playbackStore.currentTime` updates in real time during playback
3. `playbackStore.seekToTime(seconds)` seeks WaveSurfer with 50ms pre-roll
4. Selecting a word range creates a WaveSurfer Regions region over the correct time span
5. `usePlaybackStore` exposed globally for TranscriptPanel to read `currentTime` and `isWordSelected`

```typescript
import { defineStore } from 'pinia'

export const usePlaybackStore = defineStore('playback', {
  state: () => ({
    // Set by WaveformPanel after WaveSurfer initialises
    isWaveSurferReady: false,
    isPlaying: false,
    currentTime: 0,        // updated by WaveSurfer 'timeupdate' event
    duration: 0,           // updated by WaveSurfer 'ready' event

    // Word index selection (into transcriptStore.words)
    // selectionAnchor: first clicked word index
    // selectionEnd: shift-clicked word index (null = same as anchor = single word)
    selectionAnchor: null as number | null,
    selectionEnd: null as number | null,

    // Internal WaveSurfer reference — stored as non-reactive via markRaw
    _ws: null as any,
  }),

  getters: {
    hasSelection: (s): boolean => s.selectionAnchor !== null,
    selectionMin: (s): number | null => {
      if (s.selectionAnchor === null) return null
      if (s.selectionEnd === null) return s.selectionAnchor
      return Math.min(s.selectionAnchor, s.selectionEnd)
    },
    selectionMax: (s): number | null => {
      if (s.selectionAnchor === null) return null
      if (s.selectionEnd === null) return s.selectionAnchor
      return Math.max(s.selectionAnchor, s.selectionEnd)
    },
    isWordSelected(): (wordIndex: number) => boolean {
      return (wordIndex: number): boolean => {
        const min = this.selectionMin
        const max = this.selectionMax
        if (min === null || max === null) return false
        return wordIndex >= min && wordIndex <= max
      }
    },
  },

  actions: {
    registerWaveSurfer(ws: any): void {
      this._ws = ws
      this.isWaveSurferReady = false  // will be set true on 'ready' event
    },

    onWaveSurferReady(duration: number): void {
      this.isWaveSurferReady = true
      this.duration = duration
    },

    onTimeUpdate(time: number): void {
      this.currentTime = time
    },

    onPlayPause(playing: boolean): void {
      this.isPlaying = playing
    },

    // D-01: seek only (no auto-play), D-02: 50ms pre-roll
    seekToTime(timeInSeconds: number): void {
      const ws = this._ws
      if (!ws || !this.isWaveSurferReady) return
      const t = Math.max(0, timeInSeconds - 0.05)
      // setTime() is WaveSurfer v7.4+; fall back to seekTo() ratio
      if (typeof ws.setTime === 'function') {
        ws.setTime(t)
      } else if (this.duration > 0) {
        ws.seekTo(t / this.duration)
      }
    },

    // D-03/D-04: shift+click extends selection
    extendSelection(wordIndex: number): void {
      if (this.selectionAnchor === null) {
        this.selectionAnchor = wordIndex
        this.selectionEnd = wordIndex
      } else {
        this.selectionEnd = wordIndex
      }
    },

    // D-05: click without shift clears selection and sets anchor
    setAnchorOnly(wordIndex: number): void {
      this.selectionAnchor = wordIndex
      this.selectionEnd = wordIndex
    },

    clearSelection(): void {
      this.selectionAnchor = null
      this.selectionEnd = null
    },

    togglePlayPause(): void {
      this._ws?.playPause()
    },
  },
})
```

```vue
<template>
  <div class="waveform-panel">
    <!-- Placeholder when no audio loaded -->
    <div v-if="!recordingStore.audioBlobUrl" class="waveform-panel__empty">
      <p>Waveform appears after recording</p>
    </div>

    <!-- WaveSurfer container -->
    <div v-else class="waveform-panel__wave" ref="containerRef" />
  </div>
</template>

<script setup lang="ts">
import { ref, watch, onUnmounted } from 'vue'
import { useWaveSurfer } from '@meersagor/wavesurfer-vue'
import RegionsPlugin from 'wavesurfer.js/dist/plugins/regions.esm.js'
import { useRecordingStore } from '../stores/recording'
import { usePlaybackStore } from '../stores/playback'
import { useTranscriptStore } from '../stores/transcript'

const recordingStore = useRecordingStore()
const playbackStore = usePlaybackStore()
const transcriptStore = useTranscriptStore()

const containerRef = ref<HTMLElement | null>(null)

// Regions plugin instance — created before WaveSurfer init
const regionsPlugin = RegionsPlugin.create()

// WaveSurfer composable
const { waveSurfer, isReady, currentTime, isPlaying, totalDuration } = useWaveSurfer({
  containerRef,
  options: {
    height: 160,
    waveColor: '#d1d5db',
    progressColor: '#1a56db',
    cursorColor: '#374151',
    barWidth: 2,
    barGap: 1,
    barRadius: 2,
    url: recordingStore.audioBlobUrl ?? undefined,
    interact: true,
    normalize: true,
    plugins: [regionsPlugin],
  },
})

// Register WaveSurfer instance in playback store
watch(waveSurfer, (ws) => {
  if (ws) {
    playbackStore.registerWaveSurfer(ws)

    ws.on('ready', (dur: number) => {
      playbackStore.onWaveSurferReady(dur)
    })

    ws.on('timeupdate', (t: number) => {
      playbackStore.onTimeUpdate(t)
    })

    ws.on('play', () => playbackStore.onPlayPause(true))
    ws.on('pause', () => playbackStore.onPlayPause(false))
    ws.on('finish', () => playbackStore.onPlayPause(false))
  }
})

// Sync selection region with word selection from playback store
watch(
  () => [playbackStore.selectionMin, playbackStore.selectionMax] as [number | null, number | null],
  ([min, max]) => {
    regionsPlugin.clearRegions()
    if (min !== null && max !== null && isReady.value && transcriptStore.words.length > 0) {
      const startTime = transcriptStore.words[min]?.start ?? 0
      const endTime = transcriptStore.words[max]?.end ?? (startTime + 0.1)
      if (endTime > startTime) {
        regionsPlugin.addRegion({
          start: startTime,
          end: endTime,
          color: 'rgba(26, 86, 219, 0.12)',
          drag: false,
          resize: false,
        })
      }
    }
  }
)

onUnmounted(() => {
  waveSurfer.value?.destroy()
})
</script>

<style scoped>
.waveform-panel {
  height: 100%;
  display: flex;
  align-items: center;
  padding: 0 16px;
}
.waveform-panel__empty {
  color: #d1d5db;
  font-size: 0.8rem;
  letter-spacing: 0.05em;
}
.waveform-panel__wave {
  width: 100%;
  height: 100%;
}
</style>
```
