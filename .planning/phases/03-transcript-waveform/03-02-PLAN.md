# Plan 03-02: Interactive Word Spans — Click-to-Seek, Shift+Click Selection & Underline Playback Animation

## Objective

Replace the plain-text `<p>{{ transcriptStore.fullText }}</p>` in `TranscriptPanel.vue` with individual `<span>` elements per word. Each span: responds to click (seek-only, D-01) and shift+click (extend selection, D-03), applies a selection background class when its index is in the selected range, and applies an underline animation class (D-06) when it is the currently-playing word (determined by `playbackStore.currentTime`). All other TranscriptPanel states (empty, transcribing, error) remain untouched.

## Must-Haves

1. Each word is a separate clickable `<span>`
2. Clicking a word seeks WaveSurfer to `word.start - 0.05s` (seek only, stays paused)
3. Shift+clicking extends the selection; clicking alone clears and re-anchors
4. Selected words show a background tint
5. The currently-playing word shows an underline animation (no background change)

          class="word-span"

            'word-span--active': word.id === currentWordIndex,

<script setup lang="ts">
import { computed } from 'vue'
import { useTranscriptStore } from '../stores/transcript'
import { usePlaybackStore } from '../stores/playback'

const transcriptStore = useTranscriptStore()
const playbackStore = usePlaybackStore()

// Current word index — determined by comparing playbackStore.currentTime against word timestamps
// D-07: polling via WaveSurfer 'timeupdate' (driven by playbackStore.onTimeUpdate)
const currentWordIndex = computed((): number => {
  if (!transcriptStore.hasTranscript || transcriptStore.words.length === 0) return -1
  const t = playbackStore.currentTime
  // Find first word where start <= currentTime < end
  const idx = transcriptStore.words.findIndex((w) => t >= w.start && t < w.end)
  return idx  // returns -1 if not found (between words or before first)
})

// D-01: seek only — no auto-play
// D-02: 50ms pre-roll handled inside playbackStore.seekToTime()
// D-03: shift+click extends selection from anchor
// D-05: regular click clears selection and re-anchors
function handleWordClick(wordId: number, wordStart: number, event: MouseEvent): void {
  if (event.shiftKey && playbackStore.selectionAnchor !== null) {
    // Extend selection — D-03
    playbackStore.extendSelection(wordId)
  } else {
    // Seek + clear selection + set new anchor — D-01, D-05
    playbackStore.clearSelection()
    playbackStore.setAnchorOnly(wordId)
    playbackStore.seekToTime(wordStart)
  }
}

function formatDuration(seconds: number): string {
  const m = Math.floor(seconds / 60)
  const s = Math.floor(seconds % 60)
  return `${m}:${s.toString().padStart(2, '0')}`
}
</script>
