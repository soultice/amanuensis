# Plan 01-01: Electron + Vue 3 Scaffold, IPC Bridge & App Shell

## Objective

Scaffold the complete Electron + Vue 3 + TypeScript project using electron-vite. Establish the secure IPC bridge via contextBridge. Build the full app shell layout (toolbar, waveform panel, transcript panel, annotation panel, status bar) as empty but structurally complete components. Build the onboarding view with model status states and the blocking sidecar error dialog.

## Must-Haves

1. App window opens with contextIsolation=true and nodeIntegration=false
2. contextBridge exposes `window.api` with typed methods
3. Full app shell layout rendered (all panels present, empty/disabled)
4. Onboarding view shown when model is not present, with model status visible
5. Blocking sidecar error dialog prevents interaction when sidecar fails

### Task 1.3 — Implement contextBridge preload script

<read_first>
- src/preload/index.ts (current generated content — replace entirely)
- .planning/phases/01-foundation/01-RESEARCH.md §2 (full contextBridge implementation)
</read_first>

<action>
Replace `src/preload/index.ts` with:

```typescript
import { contextBridge, ipcRenderer } from 'electron'

export type ModelState = 'present' | 'downloading' | 'missing' | 'error' | 'checking'

export interface ModelStatus {
  state: ModelState
  progress?: number   // 0-100 when downloading
  error?: string
}

contextBridge.exposeInMainWorld('api', {
  // Sidecar
  getSidecarPort: (): Promise<number | null> =>
    ipcRenderer.invoke('sidecar:get-port'),
  getSidecarStatus: (): Promise<'starting' | 'ready' | 'error'> =>
    ipcRenderer.invoke('sidecar:get-status'),

  // Model
  getModelStatus: (): Promise<ModelStatus> =>
    ipcRenderer.invoke('model:get-status'),
  downloadModel: (): Promise<void> =>
    ipcRenderer.invoke('model:download'),

  // App
  restartApp: (): Promise<void> =>
    ipcRenderer.invoke('app:restart'),

  // Events
  onModelProgress: (cb: (progress: number) => void): (() => void) => {
    const handler = (_e: unknown, p: number) => cb(p)
    ipcRenderer.on('model:progress', handler)
    return () => ipcRenderer.removeListener('model:progress', handler)
  },
  onSidecarError: (cb: (err: string) => void): (() => void) => {
    const handler = (_e: unknown, e: string) => cb(e)
    ipcRenderer.on('sidecar:error', handler)
    return () => ipcRenderer.removeListener('sidecar:error', handler)
  }
})
```

Create `src/renderer/src/env.d.ts` with the Window interface:
```typescript
/// <reference types="vite/client" />

import type { ModelStatus } from '../../preload/index'

declare global {

    api: {

      getSidecarStatus(): Promise<'starting' | 'ready' | 'error'>
      getModelStatus(): Promise<ModelStatus>
      downloadModel(): Promise<void>
      restartApp(): Promise<void>
      onModelProgress(cb: (progress: number) => void): () => void
      onSidecarError(cb: (err: string) => void): () => void
    }
  }
}
```

2. Create `src/renderer/src/stores/sidecar.ts`:
```typescript
import { defineStore } from 'pinia'

export const useSidecarStore = defineStore('sidecar', {
  state: () => ({
    port: null as number | null,
    status: 'starting' as 'starting' | 'ready' | 'error',
    error: null as string | null,
  }),
  actions: {
    async initialize() {
      const port = await window.api.getSidecarPort()
      const status = await window.api.getSidecarStatus()
      this.port = port
      this.status = status
      window.api.onSidecarError((err) => {
        this.status = 'error'
        this.error = err
      })
    }
  }
})
```

3. Create `src/renderer/src/stores/model.ts`:
```typescript
import { defineStore } from 'pinia'
import type { ModelState } from '../../../../preload/index'

export const useModelStore = defineStore('model', {
  state: () => ({
    state: 'checking' as ModelState,
    progress: 0,
    error: null as string | null,
  }),
  getters: {
    isReady: (s) => s.state === 'present',
    isDownloading: (s) => s.state === 'downloading',
  },
  actions: {
    async refresh() {
      const status = await window.api.getModelStatus()
      this.state = status.state
      this.progress = status.progress ?? 0
      this.error = status.error ?? null
    },
    async startDownload() {
      this.state = 'downloading'
      const cleanup = window.api.onModelProgress((p) => {
        this.progress = p
        if (p >= 100) {
          this.state = 'present'
          cleanup()
        }
      })
      await window.api.downloadModel()
    }
  }
})
```
</action>
