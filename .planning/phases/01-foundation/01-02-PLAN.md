# Plan 01-02: Python FastAPI Sidecar with Poetry, Model Management & Sidecar Integration

## Objective

Build the Python FastAPI sidecar using Poetry for dependency management. Implement the `/health` and model management endpoints. Wire up the Electron main process to spawn the sidecar, discover its port using `get-port`, poll for health, and replace the stub IPC handlers with real implementations. Add model download with progress events. Write README setup instructions.

## Must-Haves

1. Python sidecar starts via `poetry run python -m app.main --port N --user-data PATH`
2. `GET /health` returns `{"status": "ok"}` within 30s of app start
3. Electron main process spawns sidecar on app ready, kills it on quit
4. `window.api.getSidecarPort()` returns the actual port (not null)
5. Model status correctly reflects file-system state (present/missing)
6. Model download posts progress events to renderer

```python
"""
Model manager for Whisper large-v3.
Checks presence, downloads via huggingface_hub, reports progress.
"""
from __future__ import annotations

import threading
from pathlib import Path
from typing import Callable, Literal, Optional

ModelState = Literal["present", "downloading", "missing", "error"]


class ModelManager:
    MODEL_REPO = "Systran/faster-whisper-large-v3"
    MODEL_MARKER = "model.bin"  # presence check file

    def __init__(self, user_data_dir: str) -> None:
        self._model_dir = Path(user_data_dir) / "models" / "large-v3"
        self._state: ModelState = "missing"
        self._progress: int = 0
        self._error: Optional[str] = None
        self._lock = threading.Lock()
        self._progress_callback: Optional[Callable[[int], None]] = None
        self._refresh_state()

    def _refresh_state(self) -> None:
        if (self._model_dir / self.MODEL_MARKER).exists():
            self._state = "present"
        else:
            self._state = "missing"

    @property
    def state(self) -> ModelState:
        return self._state

    @property
    def progress(self) -> int:
        return self._progress

    @property
    def error(self) -> Optional[str]:
        return self._error

    def get_model_path(self) -> Path:
        return self._model_dir

    def set_progress_callback(self, cb: Callable[[int], None]) -> None:
        self._progress_callback = cb

    def download(self) -> None:
        """Download model synchronously. Call in a background thread."""
        with self._lock:
            if self._state == "present":
                return
            self._state = "downloading"
            self._progress = 0
            self._error = None

        try:
            from huggingface_hub import snapshot_download
            self._model_dir.mkdir(parents=True, exist_ok=True)

            # snapshot_download handles resumption automatically
            snapshot_download(
                repo_id=self.MODEL_REPO,
                local_dir=str(self._model_dir),
                local_dir_use_symlinks=False,
            )

            with self._lock:
                self._state = "present"
                self._progress = 100
            if self._progress_callback:
                self._progress_callback(100)

        except Exception as exc:
            with self._lock:
                self._state = "error"
                self._error = str(exc)
```

from app.state import model_manager

router = APIRouter(prefix="/model")

@router.get("/status")
async def model_status():
    return {
        "state": model_manager.state,
        "progress": model_manager.progress,
        "error": model_manager.error,
    }

@router.post("/download")
async def start_download():
    if model_manager.state == "present":
        return {"status": "already_present"}
    if model_manager.state == "downloading":
        return {"status": "already_downloading"}
    thread = threading.Thread(target=model_manager.download, daemon=True)
    thread.start()
    return {"status": "started"}

```typescript
import { ChildProcess, spawn } from 'child_process'
import { app } from 'electron'
import path from 'path'
import getPort from 'get-port'

let sidecarProcess: ChildProcess | null = null
let sidecarPort: number | null = null
let sidecarStatus: 'starting' | 'ready' | 'error' = 'starting'
let sidecarError: string | null = null

export async function startSidecar(): Promise<number> {
  const port = await getPort()
  sidecarPort = port

  const userDataDir = app.getPath('userData')
  const isDev = !app.isPackaged

  let command: string
  let args: string[]
  let cwd: string | undefined

  if (isDev) {
    // Development: use Poetry
    const sidecarDir = path.join(process.cwd(), 'sidecar')
    command = process.platform === 'win32' ? 'poetry.exe' : 'poetry'
    args = ['run', 'python', '-m', 'app.main', '--port', String(port), '--user-data', userDataDir]
    cwd = sidecarDir
  } else {
    // Production: use PyInstaller binary
    const sidecarBin = path.join(
      process.resourcesPath,
      'sidecar',
      process.platform === 'win32' ? 'amanuensis-sidecar.exe' : 'amanuensis-sidecar'
    )
    command = sidecarBin
    args = ['--port', String(port), '--user-data', userDataDir]
  }

  sidecarProcess = spawn(command, args, {
    cwd,
    stdio: ['ignore', 'pipe', 'pipe'],
    detached: false,
  })

  sidecarProcess.stderr?.on('data', (data: Buffer) => {
    console.error('[sidecar]', data.toString())
  })

  sidecarProcess.on('exit', (code) => {
    if (code !== 0 && code !== null) {
      sidecarStatus = 'error'
      sidecarError = `Sidecar exited with code ${code}`
    }
  })

  sidecarProcess.on('error', (err) => {
    sidecarStatus = 'error'
    sidecarError = err.message
  })

  try {
    await waitForSidecar(port)
    sidecarStatus = 'ready'
  } catch (err) {
    sidecarStatus = 'error'
    sidecarError = String(err)
    throw err
  }

  return port
}

async function waitForSidecar(port: number, timeoutMs = 30000): Promise<void> {
  const start = Date.now()
  while (Date.now() - start < timeoutMs) {
    try {
      const res = await fetch(`http://localhost:${port}/health`)
      if (res.ok) return
    } catch { /* not ready yet */ }
    await new Promise(r => setTimeout(r, 250))
  }
  throw new Error(`Sidecar did not respond within ${timeoutMs}ms on port ${port}`)
}

export function getSidecarPort() { return sidecarPort }
export function getSidecarStatus() { return sidecarStatus }
export function getSidecarError() { return sidecarError }

export function killSidecar(): void {
  if (sidecarProcess) {
    sidecarProcess.kill()
    sidecarProcess = null
  }
}
```

  // Proxy to sidecar if ready, else check filesystem directly
  const port = getSidecarPort()
  const status = getSidecarStatus()
  if (status === 'ready' && port) {
    try {
      const res = await fetch(`http://localhost:${port}/model/status`)
      return await res.json()
    } catch { /* fallthrough to filesystem check */ }
  }
  // Fallback: check userData directly
  const { existsSync } = await import('fs')
  const { join } = await import('path')
  const modelBin = join(app.getPath('userData'), 'models', 'large-v3', 'model.bin')
  return { state: existsSync(modelBin) ? 'present' : 'missing' }
})

ipcMain.handle('model:download', async () => {
  const port = getSidecarPort()
  if (!port) throw new Error('Sidecar not ready')
  await fetch(`http://localhost:${port}/model/download`, { method: 'POST' })
})

ipcMain.handle('app:restart', () => {
  app.relaunch()
  app.exit(0)
})

Cross-platform desktop app for recording audio, transcribing with Whisper, annotating, and exporting Markdown documentation.

- Python >= 3.10 < 3.13
- [Poetry](https://python-poetry.org/docs/#installation) >= 1.8

npm install

cd sidecar
poetry install
cd ..

npm run dev

On first launch, the app will prompt you to download the Whisper large-v3 model (~3 GB). The download uses HuggingFace Hub and can be resumed if interrupted.
