```typescript
import { defineStore } from 'pinia'

export type RecordingState = 'idle' | 'recording' | 'stopped'

export const useRecordingStore = defineStore('recording', {
  state: () => ({
    recordingState: 'idle' as RecordingState,
    devices: [] as MediaDeviceInfo[],
    selectedDeviceId: '' as string,
    audioBlob: null as Blob | null,
    audioBlobUrl: null as string | null,
    levelBars: new Array(12).fill(0) as number[],  // 12 bars, 0-100 each
    durationMs: 0,
    error: null as string | null,
    // Internal — not reactive, managed directly
    _recorder: null as MediaRecorder | null,
    _stream: null as MediaStream | null,
    _chunks: [] as BlobPart[],
    _startTime: 0,
  }),

  getters: {
    hasRecording: (s): boolean => s.audioBlob !== null,
    isRecording: (s): boolean => s.recordingState === 'recording',
    isStopped: (s): boolean => s.recordingState === 'stopped',
  },

  actions: {
    async loadDevices(): Promise<void> {
      this.error = null
      try {
        // Must request permission first so device labels are populated
        const permStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false })
        permStream.getTracks().forEach(t => t.stop())  // release immediately

        const all = await navigator.mediaDevices.enumerateDevices()
        this.devices = all.filter(d => d.kind === 'audioinput')
        if (this.devices.length > 0 && !this.selectedDeviceId) {
          this.selectedDeviceId = this.devices[0].deviceId
        }
      } catch (err) {
        this.error = `Microphone access denied: ${String(err)}`
      }
    },

    async startRecording(): Promise<void> {
      if (this.recordingState !== 'idle' && this.recordingState !== 'stopped') return
      this.error = null
      this.audioBlob = null
      if (this.audioBlobUrl) {
        URL.revokeObjectURL(this.audioBlobUrl)
        this.audioBlobUrl = null
      }
      this._chunks = []

      try {
        const stream = await navigator.mediaDevices.getUserMedia({
          audio: {
            deviceId: this.selectedDeviceId ? { exact: this.selectedDeviceId } : undefined,
            sampleRate: 44100,
            channelCount: 1,
          }
        })
        this._stream = stream

        const mimeType = MediaRecorder.isTypeSupported('audio/webm;codecs=opus')
          ? 'audio/webm;codecs=opus'
          : 'audio/webm'
        const recorder = new MediaRecorder(stream, { mimeType })
        this._recorder = recorder

        recorder.ondataavailable = (e: BlobEvent) => {
          if (e.data.size > 0) this._chunks.push(e.data)
        }

        recorder.onstop = () => {
          const blob = new Blob(this._chunks, { type: 'audio/webm' })
          this.audioBlob = blob
          this.audioBlobUrl = URL.createObjectURL(blob)
          this.durationMs = Date.now() - this._startTime
          this.recordingState = 'stopped'
          this._stream?.getTracks().forEach(t => t.stop())
          this._stream = null
        }

        recorder.start(100)
        this._startTime = Date.now()
        this.recordingState = 'recording'
      } catch (err) {
        this.error = `Recording failed: ${String(err)}`
      }
    },

    stopRecording(): void {
      if (this._recorder && this._recorder.state === 'recording') {
        this._recorder.stop()
        // recordingState is set to 'stopped' in onstop handler
      }
    },

    setLevelBars(bars: number[]): void {
      this.levelBars = bars
    },

    resetLevelBars(): void {
      this.levelBars = new Array(12).fill(0)
    },

    clearRecording(): void {
      if (this.audioBlobUrl) URL.revokeObjectURL(this.audioBlobUrl)
      this.audioBlob = null
      this.audioBlobUrl = null
      this.durationMs = 0
      this.recordingState = 'idle'
      this.resetLevelBars()
    },
  },
})
```

<template>
  <div class="level-meter" :class="{ 'level-meter--active': props.active }">
    <div
      v-for="(height, i) in bars"
      :key="i"
      class="level-meter__bar"
      :style="{ height: `${height}%` }"
    />
  </div>
</template>

<script setup lang="ts">
import { ref, watch, onUnmounted } from 'vue'

const props = defineProps<{
  stream: MediaStream | null
  active: boolean
}>()

const BAR_COUNT = 12
const bars = ref<number[]>(new Array(BAR_COUNT).fill(0))

let audioCtx: AudioContext | null = null
let analyser: AnalyserNode | null = null
let source: MediaStreamAudioSourceNode | null = null
let rafId: number | null = null
let dataArray: Uint8Array = new Uint8Array(0)

function startMeter(stream: MediaStream): void {
  audioCtx = new AudioContext()
  analyser = audioCtx.createAnalyser()
  analyser.fftSize = 64              // 32 frequency bins
  analyser.smoothingTimeConstant = 0.8
  source = audioCtx.createMediaStreamSource(stream)
  source.connect(analyser)
  dataArray = new Uint8Array(analyser.frequencyBinCount)

  const tick = () => {
    analyser!.getByteFrequencyData(dataArray)
    const bucketSize = Math.floor(dataArray.length / BAR_COUNT)
    bars.value = Array.from({ length: BAR_COUNT }, (_, i) => {
      const start = i * bucketSize
      const slice = dataArray.slice(start, start + bucketSize)
      const avg = slice.reduce((a, b) => a + b, 0) / slice.length
      return Math.round((avg / 255) * 100)
    })
    rafId = requestAnimationFrame(tick)
  }
  rafId = requestAnimationFrame(tick)
}

function stopMeter(): void {
  if (rafId !== null) cancelAnimationFrame(rafId)
  source?.disconnect()
  audioCtx?.close()
  audioCtx = null
  analyser = null
  source = null
  rafId = null
  bars.value = new Array(BAR_COUNT).fill(0)
}

watch(
  () => [props.stream, props.active] as [MediaStream | null, boolean],
  ([stream, active]) => {
    stopMeter()
    if (stream && active) startMeter(stream)
  }
)

onUnmounted(() => stopMeter())
</script>

<script setup lang="ts">
import { ref, computed, onMounted } from 'vue'
import { useRecordingStore } from '../stores/recording'
import LevelMeter from './LevelMeter.vue'

const props = defineProps<{ isTranscribing: boolean }>()
const emit = defineEmits<{ transcribe: [] }>()

const recordingStore = useRecordingStore()
const activeStream = ref<MediaStream | null>(null)

onMounted(() => recordingStore.loadDevices())

async function startRecording(): Promise<void> {
  await recordingStore.startRecording()
