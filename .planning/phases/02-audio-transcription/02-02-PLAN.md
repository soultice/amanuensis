# Plan 02-02: Transcription Pipeline — Sidecar Endpoint, WhisperModel Singleton & Transcript Display

## Objective

Build the complete transcription pipeline: Python FastAPI `/transcribe` endpoint with WhisperModel singleton (loaded lazily on first request), word-level timestamps, `useTranscriptStore` Pinia store, and `TranscriptPanel.vue` showing inline progress while transcribing, then flowing paragraph text of the completed transcript. Wire `AppShellView.vue` to connect the Transcribe button to the pipeline.

## Must-Haves

1. `POST /transcribe` accepts WebM audio, returns `{words, language, duration}`
2. WhisperModel is loaded once and reused; not on every request
3. One transcription at a time (concurrent requests wait)
4. `TranscriptPanel` shows inline spinner + message while transcribing
5. Completed transcript renders as flowing paragraph text with word data stored
6. Language auto-detected, no user input required

```python
"""
Transcription service using faster-whisper large-v3.
WhisperModel is loaded once and reused across requests.
"""
from __future__ import annotations

import threading
from pathlib import Path
from typing import Optional, TYPE_CHECKING

if TYPE_CHECKING:
    from faster_whisper import WhisperModel  # type: ignore

_model: Optional["WhisperModel"] = None
_model_lock = threading.Lock()


def get_model(model_dir: Path) -> "WhisperModel":
    """
    Load WhisperModel once and cache as module-level singleton.
    Subsequent calls return the cached model instantly.
    Model is loaded from model_dir (the large-v3 directory in userData).
    """
    global _model
    if _model is None:
        with _model_lock:
            if _model is None:
                from faster_whisper import WhisperModel  # type: ignore

                _model = WhisperModel(
                    str(model_dir),
                    device="auto",       # GPU if CUDA available, else CPU
                    compute_type="auto", # fp16 on GPU, int8 on CPU
                    num_workers=1,
                )
    return _model


def transcribe_file(model_dir: Path, audio_path: Path) -> dict:
    """
    Transcribe a single audio file and return word-level results.
    Accepts WebM, WAV, or any format supported by PyAV (av 17+).

    Returns:
        {
            "words": [{"id": int, "word": str, "start": float, "end": float, "probability": float}],
            "language": str,
            "duration": float
        }
    """
    model = get_model(model_dir)

    segments, info = model.transcribe(
        str(audio_path),
        word_timestamps=True,
        language=None,     # auto-detect — TRANS-02
        beam_size=5,
        vad_filter=True,   # skip silence regions
    )

    words = []
    word_id = 0
    for segment in segments:
        if segment.words:
            for w in segment.words:
                words.append({
                    "id": word_id,
                    "word": w.word,
                    "start": round(w.start, 3),
                    "end": round(w.end, 3),
                    "probability": round(w.probability, 3),
                })
                word_id += 1

    return {
        "words": words,
        "language": info.language,
        "duration": round(info.duration, 2),
    }
```

```python
import asyncio
import tempfile
from pathlib import Path

from fastapi import APIRouter, UploadFile, File, HTTPException
from fastapi.responses import JSONResponse

import app.state as state
from app.services.transcription import transcribe_file

router = APIRouter(prefix="/transcribe")

# Ensure only one transcription runs at a time
_transcribe_lock = asyncio.Lock()


@router.get("/status")
async def get_transcription_status():
    """Poll this endpoint while transcription is running to show progress."""
    return JSONResponse(state.transcription_status)


@router.post("")
async def transcribe_audio(audio: UploadFile = File(...)):
    """
    Accept a WebM audio file and return word-level transcript.

    Request: multipart/form-data with field "audio" (audio/webm file)
    Response: {"words": [...], "language": str, "duration": float}

    PyAV (av 17+) handles WebM/Opus natively — no explicit conversion needed.
    Returns 503 if model is not ready.
    Returns 409 if another transcription is already running.
    """
    if state.model_manager is None or state.model_manager.state != "present":
        raise HTTPException(status_code=503, detail="Whisper model not ready")

    if _transcribe_lock.locked():
        raise HTTPException(status_code=409, detail="Transcription already in progress")

    async with _transcribe_lock:
        state.transcription_status = {"state": "running", "progress": 0, "error": None}

        tmp_path: Path | None = None
        try:
            # Write upload to temp file — faster-whisper needs a file path
            content = await audio.read()
            with tempfile.NamedTemporaryFile(suffix=".webm", delete=False) as tmp:
                tmp.write(content)
                tmp_path = Path(tmp.name)

            # Run synchronous transcription in a thread pool to avoid blocking event loop
            loop = asyncio.get_event_loop()
            model_dir = state.model_manager.get_model_path()
            result = await loop.run_in_executor(
                None, transcribe_file, model_dir, tmp_path
            )

            state.transcription_status = {"state": "complete", "progress": 100, "error": None}
            return JSONResponse(result)

        except Exception as exc:
            state.transcription_status = {
                "state": "error",
                "progress": 0,
                "error": str(exc),
            }
            raise HTTPException(status_code=500, detail=str(exc)) from exc

        finally:
            if tmp_path and tmp_path.exists():
                tmp_path.unlink(missing_ok=True)
```
</action>

```typescript
import { defineStore } from 'pinia'
import { useSidecarStore } from './sidecar'

export interface Word {
  id: number
  word: string
  start: number
  end: number
  probability: number
}

export type TranscriptState = 'empty' | 'transcribing' | 'complete' | 'error'

export const useTranscriptStore = defineStore('transcript', {
  state: () => ({
    transcriptState: 'empty' as TranscriptState,
    words: [] as Word[],
    language: '',
    duration: 0,
    error: null as string | null,
    _pollInterval: null as ReturnType<typeof setInterval> | null,
  }),

  getters: {
    // Full text for display — words joined; Phase 3 will render as individual spans
    fullText: (s): string => s.words.map(w => w.word).join(''),
    isTranscribing: (s): boolean => s.transcriptState === 'transcribing',
    hasTranscript: (s): boolean => s.transcriptState === 'complete' && s.words.length > 0,
  },

  actions: {
    async transcribe(audioBlob: Blob): Promise<void> {
      const sidecarStore = useSidecarStore()
      if (!sidecarStore.port) {
        this.error = 'Sidecar not ready'
        this.transcriptState = 'error'
        return
      }

      this.transcriptState = 'transcribing'
      this.words = []
      this.language = ''
      this.duration = 0
      this.error = null

      // Start polling /transcribe/status for UI feedback
      this._startPolling(sidecarStore.port)

      try {
        const formData = new FormData()
        formData.append('audio', audioBlob, 'recording.webm')

        const res = await fetch(`http://localhost:${sidecarStore.port}/transcribe`, {
          method: 'POST',
          body: formData,
        })

        if (!res.ok) {
          const detail = await res.text()
          throw new Error(`Transcription failed (${res.status}): ${detail}`)
        }

        const data = await res.json() as {
          words: Word[]
          language: string
          duration: number
        }

        this.words = data.words
        this.language = data.language
        this.duration = data.duration
        this.transcriptState = 'complete'

      } catch (err) {
        this.error = String(err)
        this.transcriptState = 'error'
      } finally {
        this._stopPolling()
      }
    },

    _startPolling(port: number): void {
      this._stopPolling()
      this._pollInterval = setInterval(async () => {
        try {
          await fetch(`http://localhost:${port}/transcribe/status`)
          // We don't need to parse for now — just keep the request alive
          // as a heartbeat. UI feedback is driven by transcriptState.
        } catch { /* sidecar may not be ready */ }
      }, 1000)
    },

    _stopPolling(): void {
      if (this._pollInterval !== null) {
        clearInterval(this._pollInterval)
        this._pollInterval = null
      }
    },

    reset(): void {
      this._stopPolling()
      this.transcriptState = 'empty'
      this.words = []
      this.language = ''
      this.duration = 0
      this.error = null
    },
  },
})
```
