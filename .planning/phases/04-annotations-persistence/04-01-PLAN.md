# Plan 04-01: Annotation CRUD — Store, Dialog, Floating Button, Badges, Drag-to-Move & IPC

## Objective

Build the complete annotation system: `useAnnotationStore` (Pinia) with CRUD operations and non-overlap detection; a CSS-anchored `<dialog>` (AnnotationDialog.vue) supporting image/audio file picker, clipboard paste, and text/link entry; a floating "Add annotation" button in TranscriptPanel anchored to the word selection; superscript badge numbers after annotated word ranges; AnnotationPanel listing all annotations as draggable cards; word spans accepting drops to move annotations. Wire new IPC channels for file picker, asset copy, and clipboard permission.

## Must-Haves

1. `useAnnotationStore` with CRUD + `overlapsExisting` overlap detection
2. Floating "Add annotation" button appears when word range selected, disabled if overlapping
3. CSS-anchored `<dialog>` opens adjacent to selection for create/edit (ANNOT-09)
4. Dialog supports: image from file picker (ANNOT-01), image from clipboard (ANNOT-02), text/link from clipboard (ANNOT-03), audio from file picker (ANNOT-04)
5. Annotations listed in AnnotationPanel with edit/remove (ANNOT-05, ANNOT-07)
6. Drag annotation card → drop on word span moves annotation (ANNOT-06)
7. Superscript badge numbers in transcript after annotated ranges
8. Non-overlap: word range selection blocked if overlaps existing annotation (ANNOT-08)

```typescript
import { defineStore } from 'pinia'

export type AnnotationType = 'image' | 'audio' | 'text' | 'link'

export interface Annotation {
  id: string              // UUID from crypto.randomUUID()
  wordStartIdx: number    // inclusive — index into transcriptStore.words[]
  wordEndIdx: number      // inclusive
  type: AnnotationType
  content: string         // text/URL for text+link; empty for image/audio
  assetFilename: string | null  // filename in userData/sessions/assets/ for image/audio
  createdAt: string       // ISO timestamp
}

export const useAnnotationStore = defineStore('annotation', {
  state: () => ({
    annotations: [] as Annotation[],
  }),

  getters: {
    // Sorted by wordStartIdx for display order
    sortedAnnotations(): Annotation[] {
      return [...this.annotations].sort((a, b) => a.wordStartIdx - b.wordStartIdx)
    },

    // Check if a word range [startIdx, endIdx] overlaps any existing annotation
    // excludeId: skip this annotation (for move/edit operations)
    overlapsExisting(): (startIdx: number, endIdx: number, excludeId?: string) => boolean {
      return (startIdx: number, endIdx: number, excludeId?: string): boolean =>
        this.annotations.some((a) => {
          if (a.id === excludeId) return false
          // Overlap: range A and B overlap if A.start <= B.end AND A.end >= B.start
          return startIdx <= a.wordEndIdx && endIdx >= a.wordStartIdx
        })
    },

    // Get 1-indexed badge number for the word that is the LAST word of an annotation
    // Returns null if wordIdx is not the end of any annotation
    badgeAtWord(): (wordIdx: number) => number | null {
      return (wordIdx: number): number | null => {
        const sorted = [...this.annotations].sort((a, b) => a.wordStartIdx - b.wordStartIdx)
        const idx = sorted.findIndex((a) => a.wordEndIdx === wordIdx)
        return idx >= 0 ? idx + 1 : null
      }
    },

    // Get annotation for a specific word index (for visual highlighting of annotated spans)
    annotationCoveringWord(): (wordIdx: number) => Annotation | undefined {
      return (wordIdx: number): Annotation | undefined =>
        this.annotations.find((a) => wordIdx >= a.wordStartIdx && wordIdx <= a.wordEndIdx)
    },
  },

  actions: {
    addAnnotation(annotation: Annotation): void {
      this.annotations.push(annotation)
    },

    updateAnnotation(id: string, updates: Partial<Omit<Annotation, 'id' | 'createdAt'>>): void {
      const idx = this.annotations.findIndex((a) => a.id === id)
      if (idx >= 0) Object.assign(this.annotations[idx], updates)
    },

    removeAnnotation(id: string): void {
      this.annotations = this.annotations.filter((a) => a.id !== id)
    },

    // Move annotation to a new single-word anchor (D-06/D-07)
    moveAnnotation(id: string, newWordIdx: number): void {
      const annotation = this.annotations.find((a) => a.id === id)
      if (!annotation) return
      // Collapsed to single word on drop (user can edit range in dialog afterwards)
      this.updateAnnotation(id, { wordStartIdx: newWordIdx, wordEndIdx: newWordIdx })
    },

    clearAll(): void {
      this.annotations = []
    },
  },
})
```

import { dialog } from 'electron'
import { promises as fsPromises, copyFileSync } from 'fs'
import crypto from 'crypto'

// File picker for annotation assets (images, audio clips)
ipcMain.handle('dialog:openFile', async (_event, filters: Array<{ name: string; extensions: string[] }>) => {
  const result = await dialog.showOpenDialog({
    properties: ['openFile'],
    filters,
  })
  return result.canceled ? null : result.filePaths[0] ?? null
})

// Copy a file into the session assets directory, return the new filename
ipcMain.handle('session:copyAsset', async (_event, srcPath: string) => {
  const assetsDir = path.join(app.getPath('userData'), 'sessions', 'assets')
  await fsPromises.mkdir(assetsDir, { recursive: true })
  const ext = path.extname(srcPath)
  const filename = `${crypto.randomUUID()}${ext}`
  await fsPromises.copyFile(srcPath, path.join(assetsDir, filename))
  return filename
})

// Save a blob buffer as an asset (for clipboard images)
ipcMain.handle('session:saveBlobAsset', async (_event, buffer: Buffer, ext: string) => {
  const assetsDir = path.join(app.getPath('userData'), 'sessions', 'assets')
  await fsPromises.mkdir(assetsDir, { recursive: true })
  const filename = `${crypto.randomUUID()}.${ext}`
  await fsPromises.writeFile(path.join(assetsDir, filename), buffer)
  return filename
})

// Get the assets directory path (for loading asset files)
ipcMain.handle('session:getAssetsDir', async () => {
  const assetsDir = path.join(app.getPath('userData'), 'sessions', 'assets')
  await fsPromises.mkdir(assetsDir, { recursive: true })
  return assetsDir
})

<script setup lang="ts">
import { ref, computed, onMounted } from 'vue'
import { useAnnotationStore } from '../stores/annotation'
import type { Annotation, AnnotationType } from '../stores/annotation'

const props = defineProps<{
  wordStartIdx: number
  wordEndIdx: number
  editingAnnotation?: Annotation | null
}>()
const emit = defineEmits<{ close: []; saved: [annotation: Annotation] }>()

const annotationStore = useAnnotationStore()
const dialogRef = ref<HTMLDialogElement>()

const types = [
  { value: 'image' as AnnotationType, label: 'Image' },
  { value: 'audio' as AnnotationType, label: 'Audio' },
  { value: 'text' as AnnotationType, label: 'Text' },
  { value: 'link' as AnnotationType, label: 'Link' },
]

const selectedType = ref<AnnotationType>(props.editingAnnotation?.type ?? 'image')
const textContent = ref(props.editingAnnotation?.content ?? '')
const assetFilename = ref<string | null>(props.editingAnnotation?.assetFilename ?? null)
const previewUrl = ref<string | null>(null)

const canSave = computed(() => {
  if (selectedType.value === 'image') return assetFilename.value !== null
  if (selectedType.value === 'audio') return assetFilename.value !== null
  return textContent.value.trim().length > 0
})

onMounted(() => {
  dialogRef.value?.showModal()
  // Annotation dialog IS dismissible (Escape closes it) — no preventDefault
})

async function pickImage(): Promise<void> {
  const filePath = await window.api.openFile([
    { name: 'Images', extensions: ['png', 'jpg', 'jpeg', 'gif', 'webp'] },
  ])
  if (!filePath) return
  assetFilename.value = await window.api.copyAsset(filePath)
  const assetsDir = await window.api.getAssetsDir()
  previewUrl.value = `file://${assetsDir}/${assetFilename.value}`
}

async function pasteImage(): Promise<void> {
  try {
    const items = await navigator.clipboard.read()
    for (const item of items) {
      const imageType = item.types.find((t) => t.startsWith('image/'))
      if (imageType) {
        const blob = await item.getType(imageType)
        const buffer = await blob.arrayBuffer()
        const ext = imageType.split('/')[1] ?? 'png'
        assetFilename.value = await window.api.saveBlobAsset(buffer, ext)
        const assetsDir = await window.api.getAssetsDir()
        previewUrl.value = `file://${assetsDir}/${assetFilename.value}`
        return
      }
    }
  } catch (err) {
    console.error('Clipboard read failed:', err)
  }
}

async function pickAudio(): Promise<void> {
  const filePath = await window.api.openFile([
    { name: 'Audio', extensions: ['mp3', 'wav', 'ogg', 'm4a', 'webm'] },
  ])
  if (!filePath) return
  assetFilename.value = await window.api.copyAsset(filePath)
}

async function pasteText(): Promise<void> {
  textContent.value = await navigator.clipboard.readText()
}

function save(): void {
  if (!canSave.value) return
  const annotation: Annotation = {
    id: props.editingAnnotation?.id ?? crypto.randomUUID(),
    wordStartIdx: props.wordStartIdx,
    wordEndIdx: props.wordEndIdx,
    type: selectedType.value,
    content: textContent.value.trim(),
    assetFilename: assetFilename.value,
    createdAt: props.editingAnnotation?.createdAt ?? new Date().toISOString(),
  }
  if (props.editingAnnotation) {
    annotationStore.updateAnnotation(annotation.id, annotation)
  } else {
    annotationStore.addAnnotation(annotation)
  }
  emit('saved', annotation)
  dialogRef.value?.close()
}
</script>

function handleDrop(event: DragEvent, wordIdx: number): void {
  event.preventDefault()
  dragOverWordIndex.value = null
  const annotationId = event.dataTransfer?.getData('text/plain')
  if (!annotationId) return
  // Check non-overlap: dropping onto an existing annotation's span is blocked (D-08)
