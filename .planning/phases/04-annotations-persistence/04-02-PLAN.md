# Plan 04-02: Session Persistence — Auto-Save, Save-on-Close & Restore on Relaunch

## Objective

Build `useSessionStore` managing atomic JSON auto-save every 30s (SESSION-01), save-on-close triggered by `before-quit` IPC event (SESSION-02), and session restore on relaunch (SESSION-03). Session data includes audio path, transcript words, and all annotations. Wire new IPC channels `session:save`, `session:load`, `session:triggerSave` into the main process and contextBridge. Integrate session store into AppShellView for lifecycle management.

## Must-Haves

1. `session:save` IPC writes JSON atomically: `.tmp` → `fs.rename()` (SESSION-01 atomicity)
2. Auto-save runs every 30 seconds while app is open, silent (D-09/D-10)
3. `before-quit` → main sends event → renderer saves → main allows quit (SESSION-02)
4. On relaunch, session is fully restored: transcript words, annotations, audio path (SESSION-03)
5. If audio file no longer exists, transcript and annotations still restore with a warning

import { promises as fsPromises, existsSync } from 'fs'

// Atomic session save: write to .tmp, then rename (D-13)
ipcMain.handle('session:save', async (_event, data: string) => {
  const sessionDir = path.join(app.getPath('userData'), 'sessions')
  await fsPromises.mkdir(sessionDir, { recursive: true })
  const sessionPath = path.join(sessionDir, 'current.json')
  const tmpPath = path.join(sessionDir, 'current.json.tmp')

  await fsPromises.writeFile(tmpPath, data, 'utf-8')

  // fs.rename is atomic on same filesystem; retry on Windows EPERM
  let retries = 3
  while (retries > 0) {
    try {
      await fsPromises.rename(tmpPath, sessionPath)
      break
    } catch (err: any) {
      if (err.code === 'EPERM' && retries > 1) {
        await new Promise((r) => setTimeout(r, 50))
        retries--
      } else {
        throw err
      }
    }
  }
})

// Load session — returns null if no session exists
ipcMain.handle('session:load', async () => {
  const sessionPath = path.join(app.getPath('userData'), 'sessions', 'current.json')
  try {
    return await fsPromises.readFile(sessionPath, 'utf-8')
  } catch {
    return null
  }
})

// Check if a file path exists (for audio file validation on restore — D-14)
ipcMain.handle('fs:exists', (_event, filePath: string) => {
  return existsSync(filePath)
})
```

let isQuitting = false

app.on('before-quit', async (event) => {
  if (!isQuitting) {
    event.preventDefault()
    isQuitting = true
    // Signal renderer to save synchronously, then quit
    const win = BrowserWindow.getAllWindows()[0]
    if (win) {
      win.webContents.send('app:save-and-quit')
      // Give renderer 3s to save, then force quit
      setTimeout(() => {
        killSidecar()
        app.quit()
      }, 3000)
    } else {
      killSidecar()
      app.quit()
    }
  } else {
    killSidecar()
  }
})

app.on('will-quit', () => killSidecar())
```

Add IPC handler so renderer can signal save complete:

```typescript
import { defineStore } from 'pinia'
import { useTranscriptStore } from './transcript'
import { useAnnotationStore } from './annotation'
import { useRecordingStore } from './recording'
import type { Word } from './transcript'
import type { Annotation } from './annotation'

interface SessionData {
  version: 1
  savedAt: string
  audioPath: string | null
  transcript: {
    language: string
    duration: number
    words: Word[]
  } | null
  annotations: Annotation[]
}

export const useSessionStore = defineStore('session', {
  state: () => ({
    autoSaveIntervalId: null as ReturnType<typeof setInterval> | null,
    cleanupSaveAndQuit: null as (() => void) | null,
    audioFileMissing: false,  // D-14: audio not found on restore
  }),

  actions: {
    /** Collect current state and serialize to JSON */
    buildSessionData(): string {
      const transcriptStore = useTranscriptStore()
      const annotationStore = useAnnotationStore()
      const recordingStore = useRecordingStore()

      const data: SessionData = {
        version: 1,
        savedAt: new Date().toISOString(),
        audioPath: recordingStore.audioBlobUrl
          ? recordingStore.audioBlobUrl.replace('file://', '')
          : null,
        transcript: transcriptStore.hasTranscript
          ? {
              language: transcriptStore.language,
              duration: transcriptStore.duration,
              words: transcriptStore.words,
            }
          : null,
        annotations: annotationStore.annotations,
      }
      return JSON.stringify(data)
    },

    /** Save session silently (SESSION-01, D-09) */
    async save(): Promise<void> {
      try {
        await window.api.saveSession(this.buildSessionData())
      } catch (err) {
        // Silent — do not surface errors to user (D-09)
        console.error('[session] Auto-save failed:', err)
      }
    },

    /** Restore session from disk (SESSION-03) */
    async restore(): Promise<void> {
      const raw = await window.api.loadSession()
      if (!raw) return

      try {
        const data: SessionData = JSON.parse(raw)
        const transcriptStore = useTranscriptStore()
        const annotationStore = useAnnotationStore()
        const recordingStore = useRecordingStore()

        // Restore transcript
        if (data.transcript) {
          transcriptStore.words = data.transcript.words
          transcriptStore.language = data.transcript.language
          transcriptStore.duration = data.transcript.duration
          transcriptStore.transcriptState = 'complete'
        }

        // Restore annotations
        annotationStore.annotations = data.annotations ?? []

        // Restore audio (D-14: warn if audio file missing)
        if (data.audioPath) {
          const exists = await window.api.fileExists(data.audioPath)
          if (exists) {
            recordingStore.audioBlobUrl = `file://${data.audioPath}`
            recordingStore.recordingState = 'stopped'
          } else {
            this.audioFileMissing = true  // warn in AppShellView
          }
        }
      } catch (err) {
        console.error('[session] Restore failed:', err)
      }
    },

    /** Start auto-save interval (SESSION-01: every 30s) */
    startAutoSave(): void {
      this.stopAutoSave()
      this.autoSaveIntervalId = setInterval(() => {
        this.save()
      }, 30_000)
    },

    stopAutoSave(): void {
      if (this.autoSaveIntervalId !== null) {
        clearInterval(this.autoSaveIntervalId)
        this.autoSaveIntervalId = null
      }
    },

    /** Register save-on-close handler (SESSION-02) */
    registerSaveOnClose(): void {
      this.cleanupSaveAndQuit = window.api.onSaveAndQuit(async () => {
        await this.save()
        await window.api.saveAndQuit()
      })
    },

    unregisterSaveOnClose(): void {
      this.cleanupSaveAndQuit?.()
      this.cleanupSaveAndQuit = null
    },
  },
})
```
</action>
